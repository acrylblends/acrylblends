export async function withRetry<T>(
  operation: () => Promise<T>,
  options: RetryOptions = {}
): Promise<T> {
  const {
    attempts = 4,
    baseDelay = 250,
    maxDelay = 5000,
  } = options;

  for (let attempt = 0;; attempt++) {
    try {
      return await operation();
    } catch (error) {
      if (attempt >= attempts) throw error;
      const delay = Math.min(
        baseDelay * 2 ** attempt, maxDelay
      );
      await sleep(delay);
    }
  }
}
